def detect_sun_line_hough(aopl_deg, AZIMUT_OFFSET, epsilon_deg=5):
    """
    Estime l'azimut solaire par transformée de Hough sur les pixels
    dont l'AoLP est proche de 90° (perpendiculaire au soleil).

    Entrées :
        aopl_deg       - matrice 2D de l'AoLP (degrés, 0-180)
        AZIMUT_OFFSET  - angle de l'axe X de la caméra par rapport au Nord (°)
        epsilon_deg    - demi-largeur de la bande autour de 90° (défaut 5°)

    Sorties :
        Az_sun_est     - azimut solaire estimé (degrés, 0-360)
        line_angle     - angle de la ligne détectée / horizontale (degrés)
    """

    # 1. Masque binaire : pixels dont AoLP ≈ 90°
    target = 90.0
    diff = np.abs(aopl_deg - target)
    # Gérer la circularité 0°/180°
    diff = np.minimum(diff, 180.0 - diff)
    bw = (diff < epsilon_deg).astype(np.uint8) * 255

    pct_pixels = (bw > 0).mean() * 100
    print(f"Pixels AoLP≈90° : {pct_pixels:.1f}%")

    if pct_pixels < 0.1:
        print("⚠️ Trop peu de pixels — augmenter epsilon_deg")
        return None, None

    # 2. Dilatation verticale pour relier les points
    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (1, 5))
    bw_dilated = cv2.dilate(bw, kernel)

    # 3. Transformée de Hough probabiliste
    lines = cv2.HoughLinesP(
        bw_dilated,
        rho=1,
        theta=np.pi / 180,
        threshold=30,
        minLineLength=20,
        maxLineGap=10
    )

    if lines is None:
        print("⚠️ Aucune ligne détectée par Hough")
        return None, None

    # 4. Trouver la ligne dominante (la plus longue)
    best_line = None
    best_length = 0
    for line in lines:
        x1, y1, x2, y2 = line[0]
        length = np.sqrt((x2-x1)**2 + (y2-y1)**2)
        if length > best_length:
            best_length = length
            best_line = line[0]

    x1, y1, x2, y2 = best_line
    dx = x2 - x1
    dy = y2 - y1

    # Angle de la ligne par rapport à l'horizontale
    line_angle = float(np.degrees(np.arctan2(dy, dx)) % 180.0)
    print(f"Angle ligne Hough : {line_angle:.1f}°")

    # 5. Azimut solaire — soleil perpendiculaire à la ligne
    Az_sun_est = float((line_angle + AZIMUT_OFFSET + 90.0) % 360.0)
    print(f"Azimut solaire estimé : {Az_sun_est:.1f}°")

    return Az_sun_est, line_angle


def visualiser_hough(aopl_deg, Az_sun_est, line_angle, epsilon_deg=5):
    """
    Visualise le masque AoLP≈90° et la ligne Hough détectée.
    Retourne une image BGR pour affichage Qt.
    """
    # Image AoLP normalisée en niveaux de gris
    aopl_norm = cv2.normalize(
        aopl_deg.astype(np.float32), None,
        0, 255, cv2.NORM_MINMAX
    ).astype(np.uint8)
    viz = cv2.cvtColor(aopl_norm, cv2.COLOR_GRAY2BGR)

    # Masque AoLP≈90° en rouge
    target = 90.0
    diff = np.minimum(np.abs(aopl_deg - target), 180.0 - np.abs(aopl_deg - target))
    bw = (diff < epsilon_deg).astype(np.uint8)
    viz[bw > 0] = [0, 0, 255]  # rouge

    # Ligne estimée en vert au centre
    h, w = aopl_deg.shape
    cx, cy = w // 2, h // 2
    angle_rad = np.radians(line_angle)
    length = min(h, w) // 2
    dx = int(np.cos(angle_rad) * length)
    dy = int(np.sin(angle_rad) * length)
    cv2.line(viz, (cx - dx, cy - dy), (cx + dx, cy + dy), (0, 255, 0), 2)

    # Texte
    cv2.putText(
        viz,
        f"Az={Az_sun_est:.1f}  Ligne={line_angle:.1f}",
        (10, 30),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.7, (255, 255, 255), 2
    )

    return viz

# Allow right-click saving of the image it displays
class ClickableLabel(QLabel):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.image_data = None
        self.filename = None

    def contextMenuEvent(self, event: QContextMenuEvent):
        if self.image_data is not None:
            menu = QMenu(self)
            save_action = QAction("Save image", self)
            save_action.triggered.connect(self.save_image)
            menu.addAction(save_action)
            menu.exec_(event.globalPos())

    def save_image(self):
        path, _ = QFileDialog.getSaveFileName(self, "Save image", self.filename or "capture.png", "Images (*.png *.jpg)")
        if path:
            cv2.imwrite(path, self.image_data)
            print(f"Image saved at: {path}")

    # Capture and display a frame from the live stream
    def afficher_flux(self):
        buffer = self.device.get_buffer()
        item = BufferFactory.copy(buffer)
        self.device.requeue_buffer(buffer)
        bpp = int(len(item.data) / (item.width * item.height))

        if bpp == 2:  # Mono12 is encoded on 2 bytes per pixel
            # raw data to uint6
            array = (ctypes.c_ushort * (item.width * item.height)).from_address(ctypes.addressof(item.pbytes))
            frame_16 = np.ndarray(buffer=array, dtype=np.uint16, shape=(item.height, item.width))

            # Scalling data into 8 bit format (division per 16)
            frame_8 = (frame_16 / 16).astype(np.uint8)
            qimg = QImage(frame_8.data, frame_8.shape[1], frame_8.shape[0], frame_8.strides[0], QImage.Format_Grayscale8)
            self.live_label.setPixmap(QPixmap.fromImage(qimg).scaled(1200, 800, Qt.KeepAspectRatio))

        elif bpp == 1:  # If it is Mono8 format
            array = (ctypes.c_ubyte * item.width * item.height).from_address(ctypes.addressof(item.pbytes))
            frame = np.ndarray(buffer=array, dtype=np.uint8, shape=(item.height, item.width))

            qimg = QImage(frame.data, frame.shape[1], frame.shape[0], frame.strides[0], QImage.Format_Grayscale8)
            self.live_label.setPixmap(QPixmap.fromImage(qimg).scaled(1200, 800, Qt.KeepAspectRatio))

        else:
            print(f"Unsupported BPP: {bpp}")

        BufferFactory.destroy(item)

    def auto_expose(self, target_mean=120, tolerance=15, max_iter=8):
        """
        Ajuste automatiquement l'exposition pour obtenir une mean cible.
        Retourne True si convergé, False sinon.
        """
        exposure_node = self.device.nodemap.get_node("ExposureTime")
        gain_node = self.device.nodemap.get_node("Gain")
        nodes = self.device.nodemap.get_node(['Width', 'Height', 'PixelFormat'])

        # Désactiver les autos
        auto_exp = self.device.nodemap.get_node("ExposureAuto")
        auto_gain = self.device.nodemap.get_node("GainAuto")
        if auto_exp: auto_exp.value = "Off"
        if auto_gain: auto_gain.value = "Off"

        exp_min = exposure_node.min
        exp_max = exposure_node.max
        gain_min = gain_node.min
        gain_max = gain_node.max

        # Partir d'une exposition basse en extérieur
        exposure_node.value = float(max(exp_min, 5000.0))
        gain_node.value = float(gain_min)

        for i in range(max_iter):
            # Capture rapide pour mesure
            with self.device.start_stream():
                image_buffer = self.device.get_buffer()
                raw = np.ctypeslib.as_array(
                    image_buffer.pdata,
                    shape=(image_buffer.height, image_buffer.width)
                ).astype(np.float32)
                self.device.requeue_buffer(image_buffer)
                self.device.stop_stream()

            # Mesure sur patch central uniquement
            h, w = raw.shape
            cy, cx = h // 2, w // 2
            patch = raw[cy-50:cy+50, cx-50:cx+50]
        
            # Exclure pixels saturés
            mask = patch < 200
            if mask.sum() == 0:
                current_mean = 255.0
            else:
                current_mean = patch[mask].mean()
    
            current_exp = exposure_node.value
            print(f"Iter {i+1}: expo={current_exp:.1f}µs  mean={current_mean:.1f}/201")

            # Vérifier convergence
            if abs(current_mean - target_mean) <= tolerance:
                print(f"✅ Convergé en {i+1} itérations — mean={current_mean:.1f}")
                return True

            # Saturation totale → baisser fortement
            if current_mean >= 200:
                exposure_node.value = max(exp_min, current_exp * 0.3)
                continue

            # Trop sombre → augmenter
            if current_mean < target_mean - tolerance:
                ratio = float(target_mean) / float(max(current_mean, 1.0))
                new_exp = float(min(exp_max, current_exp * ratio))
                if new_exp >= exp_max and current_mean < 30:
                    # Exposition max atteinte → monter le gain
                    new_gain = float(min(gain_max, gain_node.value + 3.0))
                    gain_node.value = new_gain
                    print(f"  → Gain augmenté à {new_gain:.1f} dB")
                else:
                    exposure_node.value = new_exp

            # Trop lumineux → baisser
            elif current_mean > target_mean + tolerance:
                ratio = float(target_mean) / float(current_mean)
                new_exp = float(max(exp_min, current_exp * ratio))
                exposure_node.value = new_exp
                # Baisser aussi le gain si possible
                if gain_node.value > gain_min:
                    gain_node.value = float(max(gain_min, gain_node.value - 3.0))

        print(f"⚠️ Non convergé après {max_iter} itérations — mean finale={current_mean:.1f}")
        return False

        
    

    # Capture a single image based on the format selected in the combo box. The supported formats are Raw Image, DoP, AOPG, and AOPL.
    def capture_image(self):
        self.arreter_previsualisation()
        QApplication.setOverrideCursor(Qt.WaitCursor)
        
        # Format selected in the drop-down list
        format_selected = self.mode_combo.currentText()
        
        #--------- THIS SECTION IS BASED ON ARENAVIEW's CODE -------  
        # === RAW IMAGE FORMAT ===
        if format_selected == "Raw Image":
            # Pixel format configuration
            tl_stream_nodemap = self.device.tl_stream_nodemap
            
            # Auto-exposition avant capture
            self.auto_expose(target_mean=30, tolerance=15, max_iter=8)

            exposure_node = self.device.nodemap.get_node("ExposureTime")
            gain_node = self.device.nodemap.get_node("Gain")
            exposure_value = exposure_node.value
            gain_value = gain_node.value
            
            
            tl_stream_nodemap['StreamAutoNegotiatePacketSize'].value = True
            tl_stream_nodemap['StreamPacketResendEnable'].value = True

            nodes = self.device.nodemap.get_node(['Width', 'Height', 'PixelFormat'])
            nodes['Width'].value = nodes['Width'].max
            nodes['Height'].value = nodes['Height'].max
            nodes['PixelFormat'].value = 'BayerRG12' 

            width = nodes['Width'].value
            height = nodes['Height'].value


       

            # Capture
            with self.device.start_stream():
                image_buffer = self.device.get_buffer()
                raw_bytes = image_buffer.data

                # Direct interpretation : every pixel = 2 bits
                image_16bit = np.frombuffer(bytes(raw_bytes), dtype=np.uint16).reshape((height, width))
                green1 = image_16bit[0::2, 1::2]
                green2 = image_16bit[1::2, 0::2]

                green = ((green1.astype(np.uint32) +
                          green2.astype(np.uint32)) // 2).astype(np.uint16)

                green_norm = cv2.normalize(
                    green.astype(np.float32), None,
                    alpha=0, beta=255,
                    norm_type=cv2.NORM_MINMAX
                ).astype(np.uint8)

                image_8bit = green_norm

               

                # Display result with PyQt
                label = ClickableLabel()
                label.setFixedSize(550, 350)

                qimg = QImage(
                    image_8bit.data,

            self.timer.stop()
            self.device.stop_stream()
        self.capture_button.setEnabled(True)

    
    # Launch a series of image captures based on GUI parameters. Then save raw image + matrice, DoP, AOPG, and computed AOPL for each step
    def lancer_acquisition(self):
        self.acquisition_start_button.setEnabled(False)
        self.acquisition_stop_button.setEnabled(True)

        # === Acquisition parameters from the GUI ===
        total_duration = self.duration_spinbox.value()
        total_unit = self.duration_unit_combobox.currentText()
        interval = self.interval_spinbox.value()
        interval_unit = self.interval_unit_combobox.currentText()
        save_path = self.path_line_edit.text()

        if save_path == "No directory selected":
            QMessageBox.warning(self, "Warning", "Please select a save directory before starting acquisition.")
            return

        # === Time conversion into seconds ===
        unit_to_seconds = {"miliseconds":0.001,"seconds": 1, "minutes": 60, "hours": 3600, "days": 86400}
        total_seconds = total_duration * unit_to_seconds[total_unit]
        interval_seconds = interval * unit_to_seconds[interval_unit]
        n_images = int(total_seconds // interval_seconds)
        
        # Counting buffers
        self.progress_bar.setMaximum(int(n_images))
        self.progress_bar.setValue(0)
        print(f"Starting acquisition for {n_images} images.")

        self.acquisition_layout.addWidget(QLabel("Acquisition started..."))
        QApplication.processEvents()

        ## CAPTURE
        try:
            # Camera settings
            # Auto-exposition avant capture
            self.auto_expose(target_mean=30, tolerance=15, max_iter=8)

            for i in range(1, n_images + 1):
                timestamp = datetime.datetime.now().strftime('%Y%m%d_%H%M%S')

            

                exposure_node = self.device.nodemap.get_node("ExposureTime")
                gain_node = self.device.nodemap.get_node("Gain")
                exposure_value = exposure_node.value
                gain_value = gain_node.value
                
                # Setup stream nodes
                self.tl_stream_nodemap['StreamAutoNegotiatePacketSize'].value = True
                self.tl_stream_nodemap['StreamPacketResendEnable'].value = True

                nodes = self.device.nodemap.get_node(['Width', 'Height', 'PixelFormat'])
                nodes['Width'].value = nodes['Width'].max
                nodes['Height'].value = nodes['Height'].max
                nodes['PixelFormat'].value = 'Mono12' 

                width = nodes['Width'].value
                height = nodes['Height'].value

                # === RAW IMAGE IN MONO12 FORMAT ===
                with self.device.start_stream():
                    image_buffer = self.device.get_buffer()
                    raw_bytes = bytes(image_buffer.data)

                    # 16 bits images (1 pixel = 2 bits in Mono12)
                    image_16bit = np.frombuffer(raw_bytes, dtype=np.uint16).reshape((height, width))

                    green1 = image_16bit[0::2, 1::2]
                    green2 = image_16bit[1::2, 0::2]

                    green = ((green1.astype(np.uint32) +
                              green2.astype(np.uint32)) // 2).astype(np.uint16)

                    green_norm = cv2.normalize(
                        green.astype(np.float32), None,
                        alpha=0, beta=255,
                        norm_type=cv2.NORM_MINMAX
                    ).astype(np.uint8)

                    image_8bit = green_norm

               

                    
                    # Save PNG 8-bit
                    raw1_filename = f"{i}_Raw_image_{timestamp}_G:{gain_value}_E:{exposure_value}.png"
                    raw_path = os.path.join(save_path, raw1_filename)
                    cv2.imwrite(raw_path, image_8bit)

                    # Save .mat with raw data 12-bit without changing anything
                    raw2_filename = f"{i}_Raw_image_{timestamp}.mat"
                    sio.savemat(os.path.join(save_path, raw2_filename), {"ImageMono12": image_16bit, "Gain_dB": gain_value, "Exposure_us": exposure_value})

                    # Display raw 8-bit image with PyQt
                    raw_label = QLabel()
                    raw_label.setFixedSize(400, 250)
                    qimg = QImage(
                        image_8bit.data,
                        image_8bit.shape[1],
                        image_8bit.shape[0],
                        image_8bit.strides[0],
                        QImage.Format_Grayscale8
                    )
                    pixmap = QPixmap.fromImage(qimg).scaled(400, 250, Qt.KeepAspectRatio)
                    raw_label.setPixmap(pixmap)

                    # Title
                    title = QLabel(f"RAW Mono12 image - number {i} - {timestamp}")
                    title.setStyleSheet("font-weight: bold; font-size: 13px")

                    # Add raw image + title 
                    self.acquisition_layout.addWidget(title)
                    self.acquisition_layout.addWidget(raw_label)

                    # Stop capture
                    self.device.requeue_buffer(image_buffer)
                    self.device.stop_stream()


                # === Display text & pregression feedback ===
                info_label = QLabel(f"✅ Raw image : {raw2_filename} saved")
                info_label.setStyleSheet("font-size: 12px; color: black")
                self.acquisition_layout.addWidget(info_label)
                
                self.progress_bar.setValue(i)
                QApplication.processEvents()
                time.sleep(interval_seconds)

        except Exception as e:
            QMessageBox.critical(self, "Acquisition Error", str(e))
            print("Exception during acquisition:", str(e))

        finally:
            self.acquisition_layout.addWidget(QLabel("Acquisition completed."))
            self.acquisition_start_button.setEnabled(True)
            self.acquisition_stop_button.setEnabled(False)
